* grouping
#+begin_src emacs-lisp
;; Don't show empty groups
(setq ibuffer-show-empty-filter-groups nil)

;; Define default groups
(setq ibuffer-saved-filter-groups
      (quote (("default"
               ("dired"    (mode . dired-mode))
               ("journal"  (name . ".jrnl.org$"))
               ("org"      (name . ".org$"))
               ("markdown" (or
                            (name . ".md$")
                            (name . "markdown$")))
               ("txt"      (name . ".txt$"))
               ("magit"    (name . "^magit"))
               ("code"     (or
			    (mode . lisp-mode)
                            (mode . python-mode)
                            (mode . c++-mode)
			    (mode . c-mode)
			    (mode . makefile-mode)))
               ("planner"  (or
                            (name . "^\\*Calendar\\*$")
                            (name . "^\\*Org Agenda\\*")))
               ("emacs"    (or
                            (name . "^\\*scratch\\*$")
                            (name . "^\\*Messages\\*$")
                            (name . "^\\*Help\\*$")))))))

(add-hook 'ibuffer-mode-hook
	  (lambda ()
	    (ibuffer-switch-to-saved-filter-groups "default")))

#+end_src
